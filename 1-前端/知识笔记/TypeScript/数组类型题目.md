## 取数组的最后一个元素的类型

错误写法（类型中可访问，不可计算）

> 1 type LastItem<T extends any[]> = T[T["length" - 1]] // 语法错误

## 初级写法

> 1 type LastItem<T extends any[]> = T extends [...any, infer Last] ? Last : never 2

> 3 type T1 = LastItem<[1, 2, 3]> // type T1 = 3

> 4 type T2 = LastItem<[]> // type T2 = never

> 5 type T3 = LastItem<number[]> // type T3 = never （不应该是 number 吗？）

高级写法

> 1 type LastItem<T extends any[]> = T extends [...any, infer Last]

> 2 ? Last

> 3 : T extends Array<infer Every>

> 4 ? Every

> 5 : never 6

> 7 type T1 = LastItem<[1, 2, 3]> // type T1 = 3

> 8 type T2 = LastItem<[]> // type T2 = never

> 9 type T3 = LastItem<number[]> // type T3 = number

## 数组的filter

错误写法

> 1 type Filter<T extends any[], F, Result extends any[] = []> = T extends [

> 2 infer L,

> 3 ...infer R

> 4 ]

> 5 ? L extends F

> 6 ? Filter<R, F, [...Result, L]>

> 7 : Filter<R, F, Result>

> 8 : Result 9

> 10 type T1 = Filter<[1, "abc", 2, true], number> // [1, 2]

> 11 type T2 = Filter<[1, "abc", 2, true], string> // ["abc"]

## 避免分发后

> 1 type Filter<T extends any[], F, Result extends any[] = []> = T extends [

> 2 infer L,

> 3 ...infer R

> 4 ]

> 5 ? [L] extends [F]

> 6 ? Filter<R, F, [...Result, L]>

> 7 : Filter<R, F, Result>

> 8 : Result 9

> 10 type T1 = Filter<[1, "abc", 2, true], number> // [1, 2]

> 11 type T2 = Filter<[1, "abc", 2, true], string> // ["abc"]

> 12 type T3 = Filter<[1, "abc", any, true], string> // ["abc", any]
